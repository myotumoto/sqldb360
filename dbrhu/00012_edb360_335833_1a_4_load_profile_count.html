WITH
by_snap AS (
SELECT /*+  NO_MERGE  */ /* 1a.2 */
       snap_id,
       metric_unit,
       metric_id,
       metric_name,
       SUM(maxval) maxval
  FROM AWR_PDB_sysmetric_summary
 WHERE snap_id BETWEEN 0 AND 1
   AND dbid = 979335833
   AND group_id =                    2
   AND metric_name LIKE '%Count'
 GROUP BY
       snap_id,
       metric_unit,
       metric_id,
       metric_name
),
by_metric_name AS (
SELECT /*+  NO_MERGE  */ /* 1a.2 */
       metric_unit,
       metric_id,
       metric_name,
       MAX(maxval) maxval_max,
       AVG(maxval) maxval_avg,

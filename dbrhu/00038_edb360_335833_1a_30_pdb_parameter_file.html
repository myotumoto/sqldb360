SELECT /*+  NO_MERGE  */ /* 1a.30 */
         pdb.pdb_id
       , pdb.pdb_name
       --, SUBSTR(pst.inst_name, 1, 9) db_name
       , UPPER(spf.db_uniq_name) db_unique_name
       , spf.sid
       , spf.name
       , spf.value$ value
       --, spf.pdb_uid
       --, pst.pdb_guid
  FROM pdb_spfile$ spf,
       pdbstate$ pst,
       dba_pdbs pdb
 WHERE pst.pdb_uid = spf.pdb_uid
   AND pdb.guid = pst.pdb_guid
 ORDER BY
       1, 2, 3, 4, 5;

    SELECT /* 2a.100 */
        di.owner,
        di.index_name,
        di.index_type,
        di.table_name,
        SUM(tm.inserts + tm.updates + tm.deletes) operations
    FROM
        cdb_indexes           di,
        cdb_tab_modifications tm
    WHERE
            di.owner in (select username from dba_users where oracle_maintained = 'N')
        AND di.table_owner NOT IN ('ANONYMOUS','APEX_030200','APEX_040000','APEX_040200','APEX_180200','APEX_SSO','APPQOSSYS','CTXSYS','DBSNMP','DIP','EXFSYS','FLOWS_FILES','MDSYS','OLAPSYS','ORACLE_OCM','ORDDATA','ORDPLUGINS','ORDSYS','OUTLN','OWBSYS')
        AND di.table_owner NOT IN ('SI_INFORMTN_SCHEMA','SQLTXADMIN','SQLTXPLAIN','SYS','SYSMAN','SYSTEM','TRCANLZR','WMSYS','XDB','XS$NULL','PERFSTAT','STDBYPERF','MGDSYS','OJVMSYS','GSMADMIN_INTERNAL')
        AND di.owner = tm.table_owner
        AND di.table_name = tm.table_name
        AND uniqueness = 'NONUNIQUE'
        AND NOT EXISTS (
            SELECT
                1
            FROM
                cdb_index_usage iu
            WHERE
                    iu.owner = di.owner
                AND iu.name = di.index_name
        )
    GROUP BY
        di.owner,
        di.index_name,
        di.index_type,
        di.table_name
    ORDER BY
        SUM(tm.inserts + tm.updates + tm.deletes) DESC,
        di.owner,
        di.table_name,
        di.index_name;

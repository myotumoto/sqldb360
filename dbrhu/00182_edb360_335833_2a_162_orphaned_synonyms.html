WITH x AS (
SELECT /*+  NO_MERGE  */ /* 2a.162 */
       s.con_id,
       s.owner, s.table_owner, COUNT(1) counter
  FROM cdb_synonyms s
 WHERE NOT EXISTS
       (SELECT NULL
          FROM cdb_objects o
         WHERE o.object_name = s.table_name
	       AND o.con_id = s.con_id
           AND o.owner = s.table_owner)
   AND s.db_link IS NULL
   AND s.owner NOT IN ('ANONYMOUS','APEX_030200','APEX_040000','APEX_040200','APEX_180200','APEX_SSO','APPQOSSYS','CTXSYS','DBSNMP','DIP','EXFSYS','FLOWS_FILES','MDSYS','OLAPSYS','ORACLE_OCM','ORDDATA','ORDPLUGINS','ORDSYS','OUTLN','OWBSYS')
   AND s.owner NOT IN ('SI_INFORMTN_SCHEMA','SQLTXADMIN','SQLTXPLAIN','SYS','SYSMAN','SYSTEM','TRCANLZR','WMSYS','XDB','XS$NULL','PERFSTAT','STDBYPERF','MGDSYS','OJVMSYS','GSMADMIN_INTERNAL')
   AND s.table_owner NOT IN ('ANONYMOUS','APEX_030200','APEX_040000','APEX_040200','APEX_180200','APEX_SSO','APPQOSSYS','CTXSYS','DBSNMP','DIP','EXFSYS','FLOWS_FILES','MDSYS','OLAPSYS','ORACLE_OCM','ORDDATA','ORDPLUGINS','ORDSYS','OUTLN','OWBSYS')
   AND s.table_owner NOT IN ('SI_INFORMTN_SCHEMA','SQLTXADMIN','SQLTXPLAIN','SYS','SYSMAN','SYSTEM','TRCANLZR','WMSYS','XDB','XS$NULL','PERFSTAT','STDBYPERF','MGDSYS','OJVMSYS','GSMADMIN_INTERNAL')
 GROUP BY
       s.con_id,
       s.owner, s.table_owner
)
 SELECT x.*
       ,c.name con_name
  FROM x
       LEFT OUTER JOIN v$containers c ON c.con_id = x.con_id
ORDER BY
       x.counter DESC;

SELECT /*+  NO_MERGE  */ /* 2b.165 */
       t.con_id,
       t.tablespace_name,
       m.used_space,
       m.tablespace_size,
       ROUND(m.used_percent, 1) used_percent,
       ROUND(m.used_space * t.block_size / POWER(2,30), 3) used_space_gb,
       ROUND(m.tablespace_size * t.block_size / POWER(2,30), 3) tablespace_size_gb
	   ,c.name con_name
  FROM cdb_tablespace_usage_metrics m,
       cdb_tablespaces t
       LEFT OUTER JOIN v$containers c ON c.con_id = t.con_id
 WHERE t.tablespace_name = m.tablespace_name
   AND t.con_id = m.con_id
 ORDER BY
       t.con_id,
       t.tablespace_name;

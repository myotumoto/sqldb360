WITH x as (
SELECT /*+  NO_MERGE  */ /* 2b.170 */
       con_id,
	   temporary_tablespace, COUNT(*) number_of_users
  FROM cdb_users
 GROUP BY
       con_id,
	   temporary_tablespace
)
SELECT x.*
       ,c.name con_name
 FROM  x
       LEFT OUTER JOIN v$containers c ON c.con_id = x.con_id
 ORDER BY
       x.con_id,
       temporary_tablespace;

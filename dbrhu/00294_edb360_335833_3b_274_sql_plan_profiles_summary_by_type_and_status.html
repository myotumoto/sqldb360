WITH x as (
SELECT COUNT(*) profiles,
       con_id,
       category,
       type,
       status,
       MIN(created) min_created,
       MAX(created) max_created,
       MEDIAN(created) median_created
  FROM cdb_sql_profiles
 GROUP BY
       con_id,
       category,
       type,
       status
)
SELECT x.*
       ,c.name con_name
  FROM x
       LEFT OUTER JOIN v$containers c ON c.con_id = x.con_id
 ORDER BY
       x.con_id,
	   x.profiles DESC, x.category, x.type, x.status;

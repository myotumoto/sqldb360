WITH x as (
SELECT con_id,
	   TO_CHAR(TRUNC(created, 'MM'), 'YYYY-MM') created,
       COUNT(*) profiles,
       SUM(CASE status WHEN 'ENABLED' THEN 1 ELSE 0 END) enabled,
       SUM(CASE status WHEN 'DISABLED' THEN 1 ELSE 0 END) disabled
  FROM cdb_sql_profiles
 GROUP BY
       con_id,
	   TRUNC(created, 'MM')
)
SELECT x.*
       ,c.name con_name
FROM   x
       LEFT OUTER JOIN v$containers c ON c.con_id = x.con_id
 ORDER BY
       x.con_id,
	   x.created;

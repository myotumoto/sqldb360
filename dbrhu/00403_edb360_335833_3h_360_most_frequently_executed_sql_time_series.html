WITH
stat AS (
SELECT /*+  MATERIALIZE NO_MERGE   DYNAMIC_SAMPLING(4)   FULL(h.ash) FULL(h.evt) FULL(h.sn) USE_HASH(h.sn h.ash h.evt)   FULL(h.INT$DBA_HIST_ACT_SESS_HISTORY.sn) FULL(h.INT$DBA_HIST_ACT_SESS_HISTORY.ash) FULL(h.INT$DBA_HIST_ACT_SESS_HISTORY.evt)   USE_HASH(h.INT$DBA_HIST_ACT_SESS_HISTORY.sn h.INT$DBA_HIST_ACT_SESS_HISTORY.ash h.INT$DBA_HIST_ACT_SESS_HISTORY.evt)  */
       /* 3h.360 */
       snap_id,
       NVL(sql_id, 'null_sql_id') sql_id,
       SUM(executions_delta) executions
  FROM AWR_PDB_sqlstat
 WHERE snap_id BETWEEN 0 AND 1
   AND dbid = 979335833
   AND NVL(sql_id, 'null_sql_id') IN
       ('not_sql_id_01',
        'not_sql_id_02',
        'not_sql_id_03',
        'not_sql_id_04',
        'not_sql_id_05',
        'not_sql_id_06',
        'not_sql_id_07',
        'not_sql_id_08',
        'not_sql_id_09',
        'not_sql_id_10',
        'not_sql_id_11',
        'not_sql_id_12',
        'not_sql_id_13',
        'not_sql_id_14',
        'not_sql_id_15')
 GROUP BY
       snap_id,
       sql_id
)
SELECT /*+  NO_MERGE  */ /* 3h.360 */
       t.snap_id,
       TO_CHAR(MIN(s.begin_interval_time), 'YYYY-MM-DD HH24:MI:SS') begin_time,
       TO_CHAR(MIN(s.end_interval_time), 'YYYY-MM-DD HH24:MI:SS') end_time,
       SUM(CASE t.sql_id WHEN 'not_sql_id_01' THEN t.executions ELSE 0 END) "not_sql_id_01",
       SUM(CASE t.sql_id WHEN 'not_sql_id_02' THEN t.executions ELSE 0 END) "not_sql_id_02",
       SUM(CASE t.sql_id WHEN 'not_sql_id_03' THEN t.executions ELSE 0 END) "not_sql_id_03",
       SUM(CASE t.sql_id WHEN 'not_sql_id_04' THEN t.executions ELSE 0 END) "not_sql_id_04",
       SUM(CASE t.sql_id WHEN 'not_sql_id_05' THEN t.executions ELSE 0 END) "not_sql_id_05",
       SUM(CASE t.sql_id WHEN 'not_sql_id_06' THEN t.executions ELSE 0 END) "not_sql_id_06",
       SUM(CASE t.sql_id WHEN 'not_sql_id_07' THEN t.executions ELSE 0 END) "not_sql_id_07",
       SUM(CASE t.sql_id WHEN 'not_sql_id_08' THEN t.executions ELSE 0 END) "not_sql_id_08",
       SUM(CASE t.sql_id WHEN 'not_sql_id_09' THEN t.executions ELSE 0 END) "not_sql_id_09",
       SUM(CASE t.sql_id WHEN 'not_sql_id_10' THEN t.executions ELSE 0 END) "not_sql_id_10",
       SUM(CASE t.sql_id WHEN 'not_sql_id_11' THEN t.executions ELSE 0 END) "not_sql_id_11",
       SUM(CASE t.sql_id WHEN 'not_sql_id_12' THEN t.executions ELSE 0 END) "not_sql_id_12",
       SUM(CASE t.sql_id WHEN 'not_sql_id_13' THEN t.executions ELSE 0 END) "not_sql_id_13",
       SUM(CASE t.sql_id WHEN 'not_sql_id_14' THEN t.executions ELSE 0 END) "not_sql_id_14",
       SUM(CASE t.sql_id WHEN 'not_sql_id_15' THEN t.executions ELSE 0 END) "not_sql_id_15"
  FROM stat t,
       AWR_PDB_snapshot s
 WHERE s.snap_id BETWEEN 0 AND 1
   AND s.dbid = 979335833
   AND t.snap_id = s.snap_id
 GROUP BY
       t.snap_id
 ORDER BY
       t.snap_id;

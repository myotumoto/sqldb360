4a.392. Total SGA Allocated (Gb) per PDB for Cluster (AWR_PDB_SGASTAT)


Memory is accounted as power of two (binary) while storage and network traffic as power of ten (decimal).
Does not include Free SGA Memory Available. For memory pools resize review Memory Statistics reports instead.

WITH
sgastat_denorm_1 AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4a.392 */
       snap_id,
       con_id,
       dbid,
       instance_number,
       SUM(bytes) sga_total
  FROM AWR_PDB_sgastat
 WHERE snap_id BETWEEN 0 AND 1
   AND dbid = 979335833
   AND instance_number = instance_number
 GROUP BY
       snap_id,
       con_id,
       dbid,
       instance_number
), sgastat_denorm_2 AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4a.392 */
       h1.snap_id,
       h1.con_id,
       h1.dbid,
       h1.instance_number,
       s1.begin_interval_time,
       s1.end_interval_time,
       ROUND((CAST(s1.end_interval_time AS DATE) - CAST(s1.begin_interval_time AS DATE)) * 24 * 60 * 60) interval_secs,
       h1.sga_total
  FROM sgastat_denorm_1 h0,
       sgastat_denorm_1 h1,
       AWR_PDB_snapshot s0,
       AWR_PDB_snapshot s1
 WHERE h1.snap_id = h0.snap_id + 1
   AND h1.dbid = h0.dbid
   and h1.con_id = h0.con_id
   AND h1.instance_number = h0.instance_number
   AND s0.snap_id = h0.snap_id
   AND s0.dbid = h0.dbid
   AND s0.instance_number = h0.instance_number
   AND s0.snap_id BETWEEN 0 AND 1
   AND s0.dbid = 979335833
   AND s1.snap_id = h1.snap_id
   AND s1.dbid = h1.dbid
   AND s1.instance_number = h1.instance_number
   AND s1.snap_id = s0.snap_id + 1
   AND s1.startup_time = s0.startup_time
   AND s1.begin_interval_time > (s0.begin_interval_time + (1 / (24 * 60))) /* filter out snaps apart < 1 min */
), x AS (
SELECT snap_id, con_id,
       TO_CHAR(MIN(begin_interval_time), 'YYYY-MM-DD HH24:MI:SS') begin_time,
       TO_CHAR(MIN(end_interval_time), 'YYYY-MM-DD HH24:MI:SS') end_time,
       ROUND(SUM(sga_total) / POWER(2,30), 3) sga_total
  FROM sgastat_denorm_2
 GROUP BY
       snap_id, con_id
), y AS (
SELECT *
  FROM x
       PIVOT(AVG(NVL(sga_total,0)) FOR con_id IN(
                   -1 DUMMY_01
,                  -1 DUMMY_02
,                  -1 DUMMY_03
,                  -1 DUMMY_04
,                  -1 DUMMY_05
,                  -1 DUMMY_06
,                  -1 DUMMY_07
,                  -1 DUMMY_08
,                  -1 DUMMY_09
,                  -1 DUMMY_10
,                  -1 DUMMY_11
,                  -1 DUMMY_12
,                  -1 DUMMY_13
,                  -1 DUMMY_14
,                  -1 DUMMY_15
))
)
select snap_id, begin_time, end_time
, NVL(DUMMY_01,0) DUMMY_01
, NVL(DUMMY_02,0) DUMMY_02
, NVL(DUMMY_03,0) DUMMY_03
, NVL(DUMMY_04,0) DUMMY_04
, NVL(DUMMY_05,0) DUMMY_05
, NVL(DUMMY_06,0) DUMMY_06
, NVL(DUMMY_07,0) DUMMY_07
, NVL(DUMMY_08,0) DUMMY_08
, NVL(DUMMY_09,0) DUMMY_09
, NVL(DUMMY_10,0) DUMMY_10
, NVL(DUMMY_11,0) DUMMY_11
, NVL(DUMMY_12,0) DUMMY_12
, NVL(DUMMY_13,0) DUMMY_13
, NVL(DUMMY_14,0) DUMMY_14
, NVL(DUMMY_15,0) DUMMY_15
from y
ORDER BY snap_id;

4h.484. "" Average Latency (AWR_PDB_EVENT_HISTOGRAM)


Trend is computed as Cumulative Weighted Average so far, weighted by the number of waits in each AWR snapshot.


Notes:
1) drag to zoom, and right click to reset
2) up to 31 days of awr history were considered

WITH
histogram AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4h.474 */
       snap_id,
       dbid,
       instance_number,
       wait_time_milli,
       (wait_count - LAG(wait_count) OVER (PARTITION BY dbid, instance_number, event_id, wait_time_milli ORDER BY snap_id)) wait_count_this_snap,
       ((CASE WHEN wait_time_milli >          .0009765625 THEN 0.75 ELSE 0.5 END)*LEAST(wait_time_milli,             8388608)) average_wait_time_milli
  FROM AWR_PDB_event_histogram
 WHERE snap_id BETWEEN 0 AND 1
   AND dbid = 979335833
   AND wait_class = TRIM('') AND event_name = TRIM('')
),
average AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4h.474 */
       snap_id,
       dbid,
       SUM(average_wait_time_milli * wait_count_this_snap) snap_wait_time_milli,
       SUM(wait_count_this_snap) wait_count_this_snap
  FROM histogram
 WHERE wait_count_this_snap >= 0
 GROUP BY
       snap_id,
       dbid
--HAVING SUM(wait_count_this_snap) > 0
),
average_average AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4h.474 */
       a.snap_id,
       a.dbid,
       MIN(a.snap_wait_time_milli/NULLIF(a.wait_count_this_snap,0)) avg_wait_time_milli,
       SUM(b.snap_wait_time_milli)/NULLIF(SUM(b.wait_count_this_snap),0) avg_avg_wait_time_milli
  FROM average a,
       average b
 WHERE b.snap_id         <= a.snap_id
   AND b.dbid             = a.dbid
 GROUP BY
       a.snap_id,
       a.dbid
),
s AS ( /*min begin/end times for each snap across all instances*/
SELECT dbid, snap_id,
       MIN(begin_interval_time) begin_interval_time,
       MIN(end_interval_time) end_interval_time
  FROM dba_hist_snapshot s
 WHERE snap_id BETWEEN 0 AND 1
   AND dbid = 979335833
GROUP BY dbid, snap_id
),
per_snap AS (
SELECT /*+  MATERIALIZE NO_MERGE  */ /* 4h.474 */
       h.snap_id,
       s.begin_interval_time,
       s.end_interval_time,
       h.avg_wait_time_milli,
       h.avg_avg_wait_time_milli
  FROM average_average h,
       s
 WHERE s.snap_id = h.snap_id
   AND s.dbid    = h.dbid
)
SELECT snap_id,
       TO_CHAR(begin_interval_time, 'YYYY-MM-DD HH24:MI:SS') begin_time,
       TO_CHAR(end_interval_time, 'YYYY-MM-DD HH24:MI:SS') end_time,
       ROUND(avg_wait_time_milli, 3) avg_latency_ms,
       ROUND(avg_avg_wait_time_milli, 3) latency_trend_ms,
       0 dummy_03,
       0 dummy_04,
       0 dummy_05,
       0 dummy_06,
       0 dummy_07,
       0 dummy_08,
       0 dummy_09,
       0 dummy_10,
       0 dummy_11,
       0 dummy_12,
       0 dummy_13,
       0 dummy_14,
       0 dummy_15
  FROM per_snap
 ORDER BY
       snap_id;

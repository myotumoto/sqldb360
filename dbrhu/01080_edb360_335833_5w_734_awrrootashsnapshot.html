SELECT *
  FROM (SELECT t.*
              ,t.snap_id-lag(t.snap_id) over (partition by con_id,dbid,instance_number order by snap_id) gap
              ,lead(t.snap_id) over (partition by con_id,dbid,instance_number order by snap_id)-t.snap_id ngap
          FROM AWR_ROOT_ASH_SNAPSHOT t
       )
 where nvl(gap,-1)<>1 or nvl(ngap,-1)<>1
 order by con_id,dbid,instance_number,snap_id;
